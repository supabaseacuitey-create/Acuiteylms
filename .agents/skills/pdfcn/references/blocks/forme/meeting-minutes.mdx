## Usage

```tsx
import { MeetingMinutesDocument } from "@/components/pdf/meeting-minutes";
```

```tsx
<MeetingMinutesDocument
  meetingTitle="Q3 Product Roadmap Review"
  date="September 12, 2026"
  time="2:00 PM - 3:30 PM"
  location="Conference Room B / Zoom"
  organizer="Alex Kim"
  attendees={[{ name: "Alex Kim", role: "Product Lead" }]}
  agenda={["Review Q2 outcomes", "Q3 feature priorities"]}
  discussions={[
    {
      topic: "Q2 Outcomes",
      notes: ["Shipped 4 of 5 planned features"],
      speaker: "Alex Kim",
    },
  ]}
  decisions={[
    {
      number: 1,
      decision: "Proceed with pdfme as second rendering base",
      rationale: "Better JSX support",
    },
  ]}
  actionItems={[
    {
      task: "Create pdfme integration RFC",
      owner: "Jordan Lee",
      dueDate: "Sep 19, 2026",
      status: "In Progress",
    },
  ]}
  preparedBy="Alex Kim"
  accentColor="#0891b2"
/>
```
